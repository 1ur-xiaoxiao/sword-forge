cmake_minimum_required(VERSION 3.16)
project(SwordForge VERSION 1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(MSVC)
    add_compile_options(/utf-8 /W4)
else()
    add_compile_options(-Wall -Wextra -Wpedantic -finput-charset=UTF-8 -fexec-charset=UTF-8)
endif()
add_library(forge_core src/data.c src/game.c src/forge.c src/adventure.c src/save.c)
target_include_directories(forge_core PUBLIC include)
add_executable(forge_game src/main.c src/ui.c)
target_link_libraries(forge_game PRIVATE forge_core)
if(WIN32)
    add_executable(sword_forge WIN32 src/web_server.c)
    target_link_libraries(sword_forge PRIVATE forge_core ws2_32 shell32)
    enable_language(RC)
    # Absolute resource paths also work when CMake builds in a nested directory.
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/assets.rc" FORGE_RESOURCES)
    string(REPLACE "\"web/" "\"${CMAKE_CURRENT_SOURCE_DIR}/web/" FORGE_RESOURCES "${FORGE_RESOURCES}")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/assets.rc" "${FORGE_RESOURCES}")
    add_executable(sword_forge_portable WIN32 src/web_server.c "${CMAKE_CURRENT_BINARY_DIR}/assets.rc")
    target_compile_definitions(sword_forge_portable PRIVATE FORGE_EMBEDDED_ASSETS)
    target_link_libraries(sword_forge_portable PRIVATE forge_core ws2_32 shell32)
    if(MINGW)
        target_link_options(sword_forge_portable PRIVATE -static -static-libgcc)
    endif()
endif()
add_executable(test_game tests/test_game.c)
target_link_libraries(test_game PRIVATE forge_core)
# Release builds still need active assertions in the test executable.
if(MSVC)
    target_compile_options(test_game PRIVATE /UNDEBUG)
else()
    target_compile_options(test_game PRIVATE -UNDEBUG)
endif()
enable_testing()
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/build")
add_test(NAME game_core COMMAND test_game)
set_tests_properties(game_core PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
